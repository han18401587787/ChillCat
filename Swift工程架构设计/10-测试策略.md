## 1. 测试金字塔

### 1.1 测试分层

```
        ╱╲
       ╱  ╲        UI Tests (E2E)      - 关键用户流程
      ╱    ╲                           - 覆盖 10-20%
     ╱──────╲
    ╱        ╲     Integration Tests    - 模块间交互
   ╱          ╲                        - 覆盖 20-30%
  ╱────────────╲
 ╱              ╲
╱────────────────╲ Unit Tests           - 核心业务逻辑
╱                  ╲                    - 覆盖 80%+
╱                    ╲
╱                      ╲
╱                        ╲
```

### 1.2 测试覆盖目标

| 层级 | 覆盖率目标 | 测试内容 |
|------|-----------|----------|
| **UseCase** | ≥ 90% | 业务逻辑、边界条件、错误路径 |
| **ViewModel** | ≥ 85% | 状态变化、用户交互、异步操作 |
| **Repository** | ≥ 80% | 数据源选择、缓存策略、错误转换 |
| **Entity/Model** | ≥ 90% | 数据校验、序列化、业务规则 |
| **View** | ≥ 70% | UI 渲染、交互响应、状态展示 |
| **Integration** | ≥ 30% | 模块间协作、数据流 |

### 2.2 UseCase 测试

```swift
// Tests/UnitTests/Domain/UseCases/LoginUseCaseTests.swift

import XCTest
@testable import ProjectName

final class LoginUseCaseTests: XCTestCase {
    private var useCase: LoginUseCase!
    private var mockRepository: MockUserRepository!

    override func setUp() {
        super.setUp()
        mockRepository = MockUserRepository()
        useCase = LoginUseCase(userRepository: mockRepository)
    }

    override func tearDown() {
        useCase = nil
        mockRepository = nil
        super.tearDown()
    }

    // MARK: - 成功路径

    func test_login_withValidCredentials_shouldReturnUser() async throws {
        // Given
        let expectedUser = User(id: "1", name: "Test User", email: "test@example.com")
        mockRepository.loginResult = .success(expectedUser)

        // When
        let user = try await useCase.execute(username: "testuser", password: "password123")

        // Then
        XCTAssertEqual(user.id, expectedUser.id)
        XCTAssertEqual(user.name, expectedUser.name)
        XCTAssertTrue(mockRepository.loginCalled)
    }

    // MARK: - 校验失败

    func test_login_withEmptyUsername_shouldThrowValidationError() async {
        // Given
        let emptyUsername = ""

        // When/Then
        do {
            _ = try await useCase.execute(username: emptyUsername, password: "password123")
            XCTFail("Expected validation error")
        } catch {
            guard let appError = error as? AppError,
                  case .validation(let validationError) = appError else {
                XCTFail("Expected validation error")
                return
            }
            XCTAssertEqual(validationError.errorDescription, "请输入用户名")
        }
    }

    func test_login_withShortPassword_shouldThrowValidationError() async {
        // Given
        let shortPassword = "123"

        // When/Then
        do {
            _ = try await useCase.execute(username: "testuser", password: shortPassword)
            XCTFail("Expected validation error")
        } catch {
            guard let appError = error as? AppError,
                  case .validation(let validationError) = appError else {
                XCTFail("Expected validation error")
                return
            }
            XCTAssertEqual(validationError.errorDescription, "密码长度不能少于 6 位")
        }
    }

    // MARK: - 业务错误

    func test_login_withInvalidCredentials_shouldThrowBusinessError() async {
        // Given
        mockRepository.loginResult = .failure(
            .business(code: 1001, message: "账号或密码错误")
        )

        // When/Then
        do {
            _ = try await useCase.execute(username: "testuser", password: "wrong")
            XCTFail("Expected business error")
        } catch {
            guard let appError = error as? AppError,
                  case .business(let code, let message) = appError else {
                XCTFail("Expected business error")
                return
            }
            XCTAssertEqual(code, 1001)
            XCTAssertEqual(message, "账号或密码错误")
        }
    }

    // MARK: - 网络错误

    func test_login_withNetworkError_shouldThrowNetworkError() async {
        // Given
        mockRepository.loginResult = .failure(.network(.networkFailure(URLError(.notConnectedToInternet))))

        // When/Then
        do {
            _ = try await useCase.execute(username: "testuser", password: "password123")
            XCTFail("Expected network error")
        } catch {
            guard let appError = error as? AppError,
                  case .network = appError else {
                XCTFail("Expected network error")
                return
            }
        }
    }
}
```

### 2.3 ViewModel 测试

```swift
// Tests/UnitTests/Presentation/ViewModels/LoginViewModelTests.swift

import XCTest
@testable import ProjectName

@MainActor
final class LoginViewModelTests: XCTestCase {
    private var viewModel: LoginViewModel!
    private var mockUseCase: MockLoginUseCase!

    override func setUp() {
        super.setUp()
        mockUseCase = MockLoginUseCase()
        viewModel = LoginViewModel(loginUseCase: mockUseCase)
    }

    override func tearDown() {
        viewModel = nil
        mockUseCase = nil
        super.tearDown()
    }

    // MARK: - 初始状态

    func test_initialState_shouldBeCorrect() {
        XCTAssertFalse(viewModel.state.isLoading)
        XCTAssertFalse(viewModel.state.isLoggedIn)
        XCTAssertNil(viewModel.state.error)
    }

    // MARK: - 登录成功

    func test_login_success_shouldUpdateState() async {
        // Given
        mockUseCase.result = .success(User(id: "1", name: "Test"))

        // When
        await viewModel.login(username: "test", password: "123456")

        // Then
        XCTAssertTrue(viewModel.state.isLoggedIn)
        XCTAssertFalse(viewModel.state.isLoading)
        XCTAssertNil(viewModel.state.error)
    }

    // MARK: - 登录失败

    func test_login_failure_shouldUpdateErrorState() async {
        // Given
        mockUseCase.result = .failure(.business(code: 1001, message: "登录失败"))

        // When
        await viewModel.login(username: "test", password: "wrong")

        // Then
        XCTAssertFalse(viewModel.state.isLoggedIn)
        XCTAssertFalse(viewModel.state.isLoading)
        XCTAssertNotNil(viewModel.state.error)
        XCTAssertEqual(viewModel.state.error?.errorDescription, "登录失败")
    }

    // MARK: - 加载状态

    func test_login_whileLoading_shouldShowLoading() async {
        // Given
        let expectation = expectation(description: "Loading state")
        mockUseCase.delay = 0.5

        // When
        Task {
            await viewModel.login(username: "test", password: "123456")
            expectation.fulfill()
        }

        // Then
        try? await Task.sleep(nanoseconds: 100_000_000) // 100ms
        XCTAssertTrue(viewModel.state.isLoading)

        await fulfillment(of: [expectation], timeout: 2)
        XCTAssertFalse(viewModel.state.isLoading)
    }
}
```

```swift
// Tests/UnitTests/Data/Repositories/UserRepositoryTests.swift

import XCTest
@testable import ProjectName

final class UserRepositoryTests: XCTestCase {
    private var repository: UserRepository!
    private var mockRemoteDataSource: MockUserRemoteDataSource!
    private var mockLocalDataSource: MockUserLocalDataSource!

    override func setUp() {
        super.setUp()
        mockRemoteDataSource = MockUserRemoteDataSource()
        mockLocalDataSource = MockUserLocalDataSource()
        repository = UserRepository(
            remoteDataSource: mockRemoteDataSource,
            localDataSource: mockLocalDataSource
        )
    }

    func test_fetchProfile_withCache_shouldReturnCachedData() async throws {
        // Given
        let cachedUser = User(id: "1", name: "Cached User")
        mockLocalDataSource.cachedUser = cachedUser

        // When
        let user = try await repository.fetchProfile()

        // Then
        XCTAssertEqual(user.id, cachedUser.id)
        XCTAssertFalse(mockRemoteDataSource.fetchProfileCalled)
    }

    func test_fetchProfile_withoutCache_shouldFetchFromNetwork() async throws {
        // Given
        let remoteUser = User(id: "2", name: "Remote User")
        mockLocalDataSource.cachedUser = nil
        mockRemoteDataSource.fetchProfileResult = .success(remoteUser)

        // When
        let user = try await repository.fetchProfile()

        // Then
        XCTAssertEqual(user.id, remoteUser.id)
        XCTAssertTrue(mockRemoteDataSource.fetchProfileCalled)
        XCTAssertTrue(mockLocalDataSource.saveUserCalled)
    }

    func test_fetchProfile_withNetworkError_shouldThrow() async {
        // Given
        mockLocalDataSource.cachedUser = nil
        mockRemoteDataSource.fetchProfileResult = .failure(.network(.notFound))

        // When/Then
        do {
            _ = try await repository.fetchProfile()
            XCTFail("Expected error")
        } catch {
            XCTAssertTrue(error is AppError)
        }
    }
}
```

```swift
// Tests/Mocks/MockUserRepository.swift

import Foundation
@testable import ProjectName

final class MockUserRepository: UserRepositoryProtocol {
    // MARK: - 调用记录
    var loginCalled = false
    var fetchProfileCalled = false
    var loginCallCount = 0

    // MARK: - 参数捕获
    var lastLoginUsername: String?
    var lastLoginPassword: String?

    // MARK: - 返回配置
    var loginResult: Result<User, AppError>?
    var fetchProfileResult: Result<User, AppError>?

    // MARK: - 延迟模拟
    var delay: TimeInterval = 0

    // MARK: - 协议实现

    func login(username: String, password: String) async throws -> User {
        loginCalled = true
        loginCallCount += 1
        lastLoginUsername = username
        lastLoginPassword = password

        if delay > 0 {
            try await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))
        }

        switch loginResult {
        case .success(let user):
            return user
        case .failure(let error):
            throw error
        case nil:
            throw AppError.unknown(nil)
        }
    }

    func fetchProfile() async throws -> User {
        fetchProfileCalled = true

        switch fetchProfileResult {
        case .success(let user):
            return user
        case .failure(let error):
            throw error
        case nil:
            throw AppError.unknown(nil)
        }
    }

    func logout() async {
        // Mock 实现
    }
}
```

## 4. 集成测试

```swift
// Tests/IntegrationTests/AuthFlowTests.swift

import XCTest
@testable import ProjectName

final class AuthFlowTests: XCTestCase {
    private var container: DIContainer!
    private var loginUseCase: LoginUseCaseProtocol!
    private var userRepository: UserRepositoryProtocol!

    override func setUp() {
        super.setUp()
        container = DIContainer()

        // 注册真实依赖，但替换网络层为 Mock
        container.register(APIClientProtocol.self) { _ in MockAPIClient() }
        container.register(KeychainManagerProtocol.self) { _ in MockKeychainManager() }
        container.register(UserDefaultsManagerProtocol.self) { _ in UserDefaultsManager() }

        // 使用真实 Repository 和 UseCase
        container.register(UserRepositoryProtocol.self) { resolver in
            UserRepository(
                remoteDataSource: UserRemoteDataSource(apiClient: resolver.resolve()),
                localDataSource: UserLocalDataSource(keychainManager: resolver.resolve())
            )
        }
        container.register(LoginUseCaseProtocol.self) { resolver in
            LoginUseCase(userRepository: resolver.resolve())
        }

        loginUseCase = container.resolve()
    }

    func test_fullLoginFlow_shouldSaveToken() async throws {
        // Given
        let mockAPI = container.resolve() as MockAPIClient
        mockAPI.mockResponse = [
            "/api/v1/auth/login": .success(LoginResponseDTO(
                accessToken: "mock_access_token",
                refreshToken: "mock_refresh_token",
                user: UserDTO(id: "1", nickName: "Test", email: "test@test.com")
            ))
        ]

        // When
        let user = try await loginUseCase.execute(username: "test", password: "password")

        // Then
        XCTAssertEqual(user.name, "Test")

        // 验证 Token 已保存
        let keychain = container.resolve() as MockKeychainManager
        let accessToken = try keychain.get("com.app.token.access")
        XCTAssertEqual(accessToken, "mock_access_token")
    }
}
```

```swift
// Tests/UITests/LoginUITests.swift

import XCTest

final class LoginUITests: XCTestCase {
    let app = XCUIApplication()

    override func setUp() {
        super.setUp()
        continueAfterFailure = false

        // 启动参数：使用 Mock 数据
        app.launchArguments = ["UI_TESTING", "-mockLogin", "true"]
        app.launch()
    }

    // MARK: - 测试用例

    func test_login_success_shouldNavigateToHome() {
        // Given
        let usernameField = app.textFields["username_field"]
        let passwordField = app.secureTextFields["password_field"]
        let loginButton = app.buttons["login_button"]

        // When
        usernameField.tap()
        usernameField.typeText("testuser")

        passwordField.tap()
        passwordField.typeText("password123")

        loginButton.tap()

        // Then
        let homeTitle = app.staticTexts["home_title"]
        XCTAssertTrue(homeTitle.waitForExistence(timeout: 5))
    }

    func test_login_emptyFields_shouldShowError() {
        // Given
        let loginButton = app.buttons["login_button"]

        // When
        loginButton.tap()

        // Then
        let errorLabel = app.staticTexts["error_message"]
        XCTAssertTrue(errorLabel.waitForExistence(timeout: 2))
        XCTAssertEqual(errorLabel.label, "请输入用户名和密码")
    }

    func test_login_invalidCredentials_shouldShowError() {
        // Given
        let usernameField = app.textFields["username_field"]
        let passwordField = app.secureTextFields["password_field"]
        let loginButton = app.buttons["login_button"]

        // When
        usernameField.tap()
        usernameField.typeText("wrong")
        passwordField.tap()
        passwordField.typeText("wrong")
        loginButton.tap()

        // Then

        XCTAssertTrue(errorAlert.waitForExistence(timeout: 3))
        XCTAssertTrue(errorAlert.staticTexts["账号或密码错误"].exists)
```

## 6. 测试最佳实践

### 6.1 测试命名规范

```swift
// 格式: test_{方法名}_{场景}_{预期结果}
func test_login_withValidCredentials_shouldReturnUser()
func test_login_withEmptyUsername_shouldThrowError()
func test_fetchFeeds_whenNetworkFails_shouldReturnCachedData()

// Given-When-Then 注释
func test_example() {
    // Given - 准备测试数据
    // When - 执行测试操作
    // Then - 验证结果
}
```

### 6.2 测试注意事项

| 注意事项 | 说明 |
|----------|------|
| **独立运行** | 每个测试用例可独立运行，不依赖其他用例 |
| **无副作用** | 测试不修改全局状态，不影响其他测试 |
| **可重复** | 多次运行结果一致 |
| **快速执行** | 单元测试应毫秒级完成 |
| **明确断言** | 每个测试用例有明确的断言 |
| **避免 sleep** | 使用 expectation 替代固定等待 |
