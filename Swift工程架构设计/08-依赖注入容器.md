## 1. 依赖注入概述

### 1.1 为什么需要 DI

| 问题 | 解决方案 |
|------|----------|
| 类之间紧耦合 | 通过协议解耦，实现依赖反转 |
| 难以测试 | 可以注入 Mock 对象 |
| 全局状态混乱 | 统一管理对象生命周期 |
| 对象创建分散 | 集中管理对象创建逻辑 |

### 1.2 DI 方式选择

| 方式 | 说明 | 推荐度 |
|------|------|--------|
| **构造函数注入** | 通过 init 传入依赖 | ⭐⭐⭐⭐⭐ |
| 属性注入 | 通过属性设置依赖 | ⭐⭐⭐ |
| 方法注入 | 通过方法参数传入 | ⭐⭐⭐ |
| 全局容器 | 通过容器解析依赖 | ⭐⭐⭐⭐ |

**推荐：** 优先使用**构造函数注入**，配合 DI 容器管理对象生命周期。

## 2. DI 容器设计

### 2.1 容器协议

```swift
// Core/DI/DIContainer.swift

import Foundation

/// 依赖注入容器协议
protocol DIContainerProtocol {
    /// 注册工厂
    func register<T>(_ type: T.Type, factory: @escaping (DIContainerProtocol) -> T)
    /// 注册具名工厂
    func register<T>(_ type: T.Type, name: String, factory: @escaping (DIContainerProtocol) -> T)
    /// 注册单例
    func registerSingleton<T>(_ type: T.Type, factory: @escaping (DIContainerProtocol) -> T)
    /// 解析依赖
    func resolve<T>() -> T
    /// 解析具名依赖
    func resolve<T>(_ name: String) -> T?
    /// 是否已注册
    func isRegistered<T>(_ type: T.Type) -> Bool
}
```

### 2.2 容器实现

```swift
/// 依赖注入容器实现
final class DIContainer: DIContainerProtocol {
    private struct Registration {
        let factory: (DIContainerProtocol) -> Any
        let isSingleton: Bool
    }

    private var registrations: [String: Registration] = [:]
    private var singletons: [String: Any] = [:]
    private let lock = NSLock()

    func register<T>(_ type: T.Type, factory: @escaping (DIContainerProtocol) -> T) {
        let key = String(describing: type)
        lock.withLock {
            registrations[key] = Registration(factory: factory, isSingleton: false)
        }
    }

    func register<T>(_ type: T.Type, name: String, factory: @escaping (DIContainerProtocol) -> T) {
        let key = "\(String(describing: type))_\(name)"
        lock.withLock {
            registrations[key] = Registration(factory: factory, isSingleton: false)
        }
    }

    func registerSingleton<T>(_ type: T.Type, factory: @escaping (DIContainerProtocol) -> T) {
        let key = String(describing: type)
        lock.withLock {
            registrations[key] = Registration(factory: factory, isSingleton: true)
        }
    }

    func resolve<T>() -> T {
        let key = String(describing: T.self)

        return lock.withLock {
            // 检查单例缓存
            if let singleton = singletons[key] as? T {
                return singleton
            }

            guard let registration = registrations[key] else {
                fatalError("❌ DI: \(key) 未注册，请先调用 register()")
            }

            let instance = registration.factory(self) as! T

            // 如果是单例，缓存起来
            if registration.isSingleton {
                singletons[key] = instance
            }

            return instance
        }
    }

    func resolve<T>(_ name: String) -> T? {
        let key = "\(String(describing: T.self))_\(name)"
        return lock.withLock {
            guard let registration = registrations[key] else { return nil }
            return registration.factory(self) as? T
        }
    }

    func isRegistered<T>(_ type: T.Type) -> Bool {
        let key = String(describing: type)
        return lock.withLock { registrations[key] != nil }
    }
}
```

### 2.3 @Inject 属性包装器

```swift
// Core/DI/Inject.swift

import Foundation

/// 属性注入包装器
@propertyWrapper
struct Inject<T> {
    private let container: DIContainerProtocol
    private let name: String?

    var wrappedValue: T {
        if let name = name, let resolved: T = container.resolve(name) {
            return resolved
        }
        return container.resolve()
    }

    init(container: DIContainerProtocol = DIContainer.global, name: String? = nil) {
        self.container = container
        self.name = name
    }
}

// 使用示例
final class SomeViewModel {
    @Inject private var userRepository: UserRepositoryProtocol
    @Inject(name: "mock") private var mockRepository: UserRepositoryProtocol?
}
```

## 3. 容器组装

### 3.1 组装器

```swift
// App/AppDependencyContainer.swift

import Foundation

/// 应用依赖容器组装器
final class AppDependencyContainer {
    static let shared = AppDependencyContainer()
    let container: DIContainer

    private init() {
        container = DIContainer()
        registerDependencies()
    }

    private func registerDependencies() {
        registerCoreServices()
        registerRepositories()
        registerUseCases()
        registerCoordinators()
    }

    // MARK: - Core Services

    private func registerCoreServices() {
        // 网络层
        container.registerSingleton(APIClientProtocol.self) { resolver in
            APIClient(
                interceptor: resolver.resolve() as TokenInterceptor,
                logger: NetworkLogger()
            )
        }

        container.registerSingleton(TokenInterceptor.self) { resolver in
            TokenInterceptor(tokenProvider: resolver.resolve())
        }

        container.registerSingleton(TokenProviderProtocol.self) { _ in
            TokenProvider(keychainManager: KeychainManager())
        }

        // 存储层
        container.registerSingleton(KeychainManagerProtocol.self) { _ in
            KeychainManager()
        }

        container.registerSingleton(UserDefaultsManagerProtocol.self) { _ in
            UserDefaultsManager()
        }

        container.registerSingleton(CacheManagerProtocol.self) { _ in
            CacheManager(cacheName: "app_cache")
        }

        // 日志
        container.registerSingleton(LoggerProtocol.self) { _ in
            Logger()
        }
    }

    // MARK: - Repositories

    private func registerRepositories() {
        container.register(UserRepositoryProtocol.self) { resolver in
            UserRepository(
                remoteDataSource: UserRemoteDataSource(
                    apiClient: resolver.resolve()
                ),
                localDataSource: UserLocalDataSource(
                    keychainManager: resolver.resolve()
                )
            )
        }

        container.register(FeedRepositoryProtocol.self) { resolver in
            FeedRepository(
                remoteDataSource: FeedRemoteDataSource(
                    apiClient: resolver.resolve()
                ),
                localDataSource: FeedLocalDataSource(
                    cacheManager: resolver.resolve()
                )
            )
        }
    }

    // MARK: - UseCases

    private func registerUseCases() {
        container.register(LoginUseCaseProtocol.self) { resolver in
            LoginUseCase(userRepository: resolver.resolve())
        }

        container.register(FetchFeedsUseCaseProtocol.self) { resolver in
            FetchFeedsUseCase(feedRepository: resolver.resolve())
        }

        container.register(LogoutUseCaseProtocol.self) { resolver in
            LogoutUseCase(userRepository: resolver.resolve())
        }
    }

    // MARK: - Coordinators

    private func registerCoordinators() {
        container.registerSingleton(AppCoordinator.self) { resolver in
            AppCoordinator(diContainer: resolver)
        }
    }
}

// 全局容器便捷访问
extension DIContainer {
    static let global: DIContainer = AppDependencyContainer.shared.container
}
```

### 3.2 模块级容器

```swift
// Modules/HomeModule/Sources/HomeDIContainer.swift

import Foundation

/// Home 模块依赖容器
final class HomeDIContainer {
    private let parentContainer: DIContainerProtocol
    private let childContainer: DIContainer

    init(parent: DIContainerProtocol = DIContainer.global) {
        self.parentContainer = parent
        self.childContainer = DIContainer()
        registerModuleDependencies()
    }

    private func registerModuleDependencies() {
        // 模块内 ViewModel
        childContainer.register(HomeViewModel.self) { resolver in
            HomeViewModel(
                fetchFeedsUseCase: resolver.resolve(),
                coordinator: self.parentContainer.resolve()
            )
        }
    }

    func resolve<T>() -> T {
        childContainer.resolve()
    }
}
```

## 4. 使用 Factory 三方库（可选）

### 4.1 Factory 配置

```swift
// 如果使用 Factory 三方库

import Factory

// 定义容器
extension Container {
    // Core Services
    var apiClient: Factory<APIClientProtocol> {
        self { APIClient(interceptor: self.tokenInterceptor()) }
            .singleton
    }

    var tokenInterceptor: Factory<TokenInterceptor> {
        self { TokenInterceptor(tokenProvider: self.tokenProvider()) }
            .singleton
    }

    var tokenProvider: Factory<TokenProviderProtocol> {
        self { TokenProvider(keychainManager: self.keychainManager()) }
            .singleton
    }

    var keychainManager: Factory<KeychainManagerProtocol> {
        self { KeychainManager() }
            .singleton
    }

    var cacheManager: Factory<CacheManagerProtocol> {
        self { CacheManager(cacheName: "app_cache") }
            .singleton
    }

    // Repositories
    var userRepository: Factory<UserRepositoryProtocol> {
        self { UserRepository(
            remoteDataSource: self.userRemoteDataSource(),
            localDataSource: self.userLocalDataSource()
        )}
    }

    var userRemoteDataSource: Factory<UserRemoteDataSource> {
        self { UserRemoteDataSource(apiClient: self.apiClient()) }
    }

    var userLocalDataSource: Factory<UserLocalDataSource> {
        self { UserLocalDataSource(keychainManager: self.keychainManager()) }
    }

    // UseCases
    var loginUseCase: Factory<LoginUseCaseProtocol> {
        self { LoginUseCase(userRepository: self.userRepository()) }
    }

    // ViewModels
    var loginViewModel: Factory<LoginViewModel> {
        self { LoginViewModel(loginUseCase: self.loginUseCase()) }
    }
}

// 使用
final class LoginView: View {
    @Injected(Container.loginViewModel) private var viewModel
    // ...
}
```

## 5. DI 最佳实践

### 5.1 生命周期管理

| 生命周期 | 注册方式 | 说明 |
|----------|----------|------|
| **单例** | `registerSingleton` | 全局唯一实例，如 APIClient |
| **每次创建** | `register` | 每次 resolve 创建新实例，如 ViewModel |
| **作用域** | 子容器 | 模块级容器，如 HomeDIContainer |

### 5.2 测试 Mock 注入

```swift
// Tests/Helpers/MockDIContainer.swift

import Foundation

/// 测试用 Mock DI 容器
final class MockDIContainer {
```

### 5.3 DI 规范

```swift
// ✅ 正确：构造函数注入
final class LoginViewModel {
    private let loginUseCase: LoginUseCaseProtocol

    init(loginUseCase: LoginUseCaseProtocol) {
        self.loginUseCase = loginUseCase
    }
}

// ❌ 错误：直接在内部创建依赖
final class LoginViewModel {
    private let loginUseCase = LoginUseCase(
        userRepository: UserRepository(
            remoteDataSource: UserRemoteDataSource(
                apiClient: APIClient()
            ),
            localDataSource: UserLocalDataSource(
                keychainManager: KeychainManager()
            )
        )
    )
}

// ✅ 正确：通过容器解析
let viewModel: LoginViewModel = container.resolve()

// ❌ 错误：使用 Service Locator 反模式
let viewModel = LoginViewModel(
    loginUseCase: ServiceLocator.shared.getLoginUseCase()
)
```

### 5.4 注册检查清单

- [ ] 所有协议都有对应的注册
- [ ] 单例服务使用 `registerSingleton`
- [ ] ViewModel 使用 `register`（非单例）
- [ ] 模块容器正确继承父容器
- [ ] 测试容器正确注册 Mock
- [ ] 没有循环依赖（A→B→A）
- [ ] 没有未注册的依赖导致运行时崩溃
