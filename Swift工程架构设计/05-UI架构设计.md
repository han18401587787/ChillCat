## 1. UI 架构总览

### 1.1 架构模式

采用 **SwiftUI + MVVM + Coordinator** 架构：

```
┌──────────────────────────────────────────────────────┐
│                    Coordinator                        │
│              (路由导航、模块间跳转)                      │
└───┬──────────────────────────────────────────────┬───┘
    │                                              │
    ▼                                              ▼
┌───────────┐    ┌───────────┐    ┌──────────────┐
│   View    │◄───│ ViewModel │◄───│   UseCase    │
│ (SwiftUI) │    │(@State)   │    │  (Domain)    │
│  只渲染    │    │  状态管理  │    │  业务逻辑     │
└───────────┘    └───────────┘    └──────────────┘
```

### 1.2 数据流

```
用户操作 → View → ViewModel.action() → UseCase.execute()
                                              │
                                              ▼
View 更新 ← ViewModel.@Published ← Repository
```

**单向数据流：** View → ViewModel → UseCase → Repository → ViewModel → View

## 2. ViewModel 设计

### 2.1 ViewModel 规范

```swift
// ✅ ViewModel 标准模板

import Foundation

// MARK: - Protocol
protocol LoginViewModelProtocol: AnyObject {
    var state: LoginState { get }
    func login(username: String, password: String) async
}

// MARK: - State
struct LoginState {
    var isLoading = false
    var error: AppError?
    var isLoggedIn = false
    var username = ""
    var password = ""
    var isFormValid: Bool {
        !username.isEmpty && !password.isEmpty && password.count >= 6
    }
}

// MARK: - ViewModel
@MainActor
@Observable
final class LoginViewModel: LoginViewModelProtocol {
    // MARK: - State
    var state = LoginState()

    // MARK: - Dependencies
    private let loginUseCase: LoginUseCaseProtocol

    // MARK: - Init
    init(loginUseCase: LoginUseCaseProtocol) {
        self.loginUseCase = loginUseCase
    }

    // MARK: - Actions
    func login(username: String, password: String) async {
        state.isLoading = true
        state.error = nil

        do {
            let user = try await loginUseCase.execute(
                username: username,
                password: password
            )
            state.isLoggedIn = true
        } catch {
            state.error = AppError.from(error)
        }

        state.isLoading = false
    }
}
```

### 2.2 ViewModel 设计原则

| 原则 | 说明 |
|------|------|
| **无 UI 依赖** | ViewModel 不 import SwiftUI/UIKit |
| **单一职责** | 一个 ViewModel 只负责一个视图的状态 |
| **可测试** | 所有依赖通过协议注入，便于 Mock |
| **主线程安全** | ViewModel 的状态更新应在 @MainActor 下 |
| **状态不可变** | 使用 struct 作为 State，每次创建新实例 |

### 2.3 ViewModel 状态管理

#### iOS 17+（推荐）

```swift
@Observable
@MainActor
final class HomeViewModel {
    var items: [Item] = []
    var isLoading = false
    var error: Error?
}
```

#### iOS 15-16（兼容）

```swift
final class HomeViewModel: ObservableObject {
    @Published var items: [Item] = []
    @Published var isLoading = false
    @Published var error: Error?
}
```

## 3. View 设计

### 3.1 View 规范

```swift
// ✅ View 标准模板

import SwiftUI

struct LoginView: View {
    // MARK: - ViewModel
    @State private var viewModel: LoginViewModel
    // 或 @StateObject（iOS 16-）
    // @StateObject private var viewModel: LoginViewModel

    // MARK: - Coordinator
    @EnvironmentObject private var coordinator: AppCoordinator

    // MARK: - Body
    var body: some View {
        content
            .task {
                // 初始化加载
            }
            .navigationTitle("登录")
    }

    // MARK: - Content
    @ViewBuilder
    private var content: some View {
        if viewModel.state.isLoading {
            LoadingView()
        } else {
            loginForm
        }
    }

    // MARK: - Subviews
    private var loginForm: some View {
        VStack(spacing: 16) {
            TextField("用户名", text: $viewModel.state.username)
                .textFieldStyle(.roundedBorder)
                .accessibilityIdentifier("username_field")

            SecureField("密码", text: $viewModel.state.password)
                .textFieldStyle(.roundedBorder)
                .accessibilityIdentifier("password_field")

            Button("登录") {
                Task {
                    await viewModel.login()
                }
            }
            .disabled(!viewModel.state.isFormValid)
            .accessibilityIdentifier("login_button")

            if let error = viewModel.state.error {
                Text(error.localizedDescription)
                    .foregroundColor(.red)
                    .accessibilityIdentifier("error_message")
            }
        }
        .padding()
    }
}
```

### 3.2 View 设计原则

| 原则 | 说明 |
|------|------|
| **纯展示** | View 只做渲染和事件转发，不做业务逻辑 |
| **细粒度组件** | 将复杂视图拆分为小组件 |
| **Accessibility** | 所有交互元素添加 accessibilityIdentifier |
| **Preview 支持** | 提供 PreviewProvider 或 #Preview |
| **状态驱动** | 视图完全由 ViewModel 的状态驱动 |

## 4. Coordinator / 路由设计

### 4.1 Coordinator 模式

```swift
// Presentation/Navigation/AppCoordinator.swift

import SwiftUI

/// 路由定义
enum AppRoute: Hashable {
    case login
    case home
    case profile
    case detail(id: String)
    case settings
    case web(url: URL)

    func hash(into hasher: inout Hasher) {
        switch self {
        case .detail(let id):
            hasher.combine("detail")
            hasher.combine(id)
        case .web(let url):
            hasher.combine("web")
            hasher.combine(url)
        default:
            hasher.combine(String(describing: self))
        }
    }
}

/// 根协调器
final class AppCoordinator: ObservableObject {
    @Published var path = NavigationPath()
    @Published var presentedSheet: AppRoute?
    @Published var presentedFullScreen: AppRoute?

    private let diContainer: DIContainerProtocol

    init(diContainer: DIContainerProtocol) {
        self.diContainer = diContainer
    }

    // MARK: - Navigation Actions

    func navigate(to route: AppRoute) {
        path.append(route)
    }

    func presentSheet(_ route: AppRoute) {
        presentedSheet = route
    }

    func presentFullScreen(_ route: AppRoute) {
        presentedFullScreen = route
    }

    func dismiss() {
        if presentedSheet != nil {
            presentedSheet = nil
        } else if presentedFullScreen != nil {
            presentedFullScreen = nil
        }
    }

    func popToRoot() {
        path.removeLast(path.count)
    }

    func pop() {
        guard !path.isEmpty else { return }
        path.removeLast()
    }

    // MARK: - View Builder

    @ViewBuilder
    func buildView(for route: AppRoute) -> some View {
        switch route {
        case .login:
            let vm = LoginViewModel(
                loginUseCase: diContainer.resolve()
            )
            LoginView(viewModel: vm)

        case .home:
            let vm = HomeViewModel(
                fetchItemsUseCase: diContainer.resolve()
            )
            HomeView(viewModel: vm)

        case .profile:
            let vm = ProfileViewModel(
                userRepository: diContainer.resolve()
            )
            ProfileView(viewModel: vm)

        case .detail(let id):
            let vm = DetailViewModel(
                itemId: id,
                fetchDetailUseCase: diContainer.resolve()
            )
            DetailView(viewModel: vm)

        case .settings:
            SettingsView()

        case .web(let url):
            WebView(url: url)
        }
    }

    func buildRootView() -> some View {
        Group {
            if isLoggedIn {
                MainTabView()
            } else {
                buildView(for: .login)
            }
        }
    }

    private var isLoggedIn: Bool {
        // 从 TokenProvider 获取登录状态
        (try? diContainer.resolve() as TokenProviderProtocol)
            .flatMap { try? $0.getAccessToken() } != nil
    }
}
```

### 4.2 导航实现

```swift
// App.swift - 集成 NavigationStack

@main
struct ProjectApp: App {
    @StateObject private var coordinator: AppCoordinator

    init() {
        let diContainer = AppDependencyContainer()
        _coordinator = StateObject(
            wrappedValue: AppCoordinator(diContainer: diContainer)
        )
    }

    var body: some Scene {
        WindowGroup {
            NavigationStack(path: $coordinator.path) {
                coordinator.buildRootView()
                    .navigationDestination(for: AppRoute.self) { route in
                        coordinator.buildView(for: route)
                    }
                    .sheet(item: $coordinator.presentedSheet) { route in
                        coordinator.buildView(for: route)
                    }
                    .fullScreenCover(item: $coordinator.presentedFullScreen) { route in
                        coordinator.buildView(for: route)
                    }
            }
            .environmentObject(coordinator)
        }
    }
}

// 使 AppRoute 支持 Identifiable
extension AppRoute: Identifiable {
    var id: String {
        switch self {
        case .detail(let id): return "detail_\(id)"
        case .web(let url): return "web_\(url.absoluteString)"
        default: return String(describing: self)
        }
    }
}
```

### 4.3 深度链接处理

```swift
// Presentation/Navigation/DeepLinkHandler.swift

import Foundation

final class DeepLinkHandler {
    private let coordinator: AppCoordinator

    init(coordinator: AppCoordinator) {
        self.coordinator = coordinator
    }

    func handle(url: URL) -> Bool {
        guard let components = URLComponents(url: url, resolvingAgainstBaseURL: false) else {
            return false
        }

        let path = components.path
        let queryItems = components.queryItems

        switch path {
        case "/detail":
            if let id = queryItems?.first(where: { $0.name == "id" })?.value {
                coordinator.navigate(to: .detail(id: id))
                return true
            }
        case "/profile":
            coordinator.navigate(to: .profile)
            return true
        case "/web":
            if let urlString = queryItems?.first(where: { $0.name == "url" })?.value,
               let url = URL(string: urlString) {
                coordinator.navigate(to: .web(url: url))
                return true
            }
        default:
            break
        }

        return false
    }
}
```

### 5.3 主题注入

```swift
// MARK: - Environment Key

struct AppThemeKey: EnvironmentKey {
    static let defaultValue: AppThemeProtocol = LightTheme()
}

extension EnvironmentValues {
    var appTheme: AppThemeProtocol {
        get { self[AppThemeKey.self] }
        set { self[AppThemeKey.self] = newValue }
    }
}

// MARK: - 主题管理器

@MainActor
@Observable
final class ThemeManager {
    var currentTheme: AppThemeProtocol = LightTheme() {
        didSet { applyTheme() }
    }

    var isDarkMode: Bool {
        get { currentTheme is DarkTheme }
        set { currentTheme = newValue ? DarkTheme() : LightTheme() }
    }

    private func applyTheme() {
        // 更新全局 UI 样式
        UIView.appearance().tintColor = UIColor(currentTheme.primary)
    }
}

// MARK: - 使用示例

struct SomeView: View {
    @Environment(\.appTheme) private var theme
    @Environment(ThemeManager.self) private var themeManager

    var body: some View {
        VStack(spacing: theme.spacingMD) {
            Text("标题")
                .font(theme.title)
                .foregroundColor(theme.textPrimary)

            Text("正文内容")
                .font(theme.body)
                .foregroundColor(theme.textSecondary)
        }
        .padding(theme.spacingLG)
        .background(theme.surface)
        .cornerRadius(theme.radiusMD)
        .shadow(
            color: theme.shadowMD.color,
            radius: theme.shadowMD.radius,
            x: theme.shadowMD.x,
            y: theme.shadowMD.y
        )
    }
}
```
