## 1. 持久化策略总览

### 1.1 存储方案选择

| 数据类型 | 存储方式 | 适用场景 | 示例 |
|----------|----------|----------|------|
| **敏感数据** | Keychain | Token、密码、密钥 | 访问令牌、刷新令牌 |
| **用户偏好** | UserDefaults / @AppStorage | 简单配置项 | 主题设置、语言偏好 |
| **结构化数据** | SwiftData / CoreData | 业务数据持久化 | 用户信息、收藏列表 |
| **临时缓存** | NSCache / 文件缓存 | 内存缓存 | 图片缓存、API 响应缓存 |
| **大文件** | FileManager | 图片、视频、文档 | 下载文件、离线资源 |
| **SQLite** | GRDB / FMDB | 复杂查询场景 | 聊天记录、离线数据 |

### 1.2 数据流决策

```
数据请求
    │
    ▼
检查内存缓存（NSCache）
    │
    ├── 命中 → 返回
    │
    ▼
检查磁盘缓存（File Cache）
    │
    ├── 命中且未过期 → 返回 + 写入内存缓存
    │
    ▼
检查本地数据库（SwiftData/CoreData）
    │
    ├── 有数据 → 返回 + 后台刷新
    │
    ▼
请求远程 API
    │
    ├── 成功 → 写入缓存 + 写入数据库 + 返回
    │
    └── 失败 → 返回错误 / 返回过期数据
```

```swift
// Core/Storage/KeychainManager.swift

import Foundation
import Security

/// Keychain 管理器协议
protocol KeychainManagerProtocol {
    func set(_ value: String, for key: String) throws
    func set(_ data: Data, for key: String) throws
    func get(_ key: String) throws -> String?
    func getData(_ key: String) throws -> Data?
    func delete(_ key: String) throws
    func clear() throws
}

/// Keychain 管理器实现
final class KeychainManager: KeychainManagerProtocol {
    private let service: String
    private let accessGroup: String?

    init(service: String = Bundle.main.bundleIdentifier ?? "com.app.keychain",
         accessGroup: String? = nil) {
        self.service = service
        self.accessGroup = accessGroup
    }

    func set(_ value: String, for key: String) throws {
        guard let data = value.data(using: .utf8) else {
            throw KeychainError.encodingFailed
        }
        try set(data, for: key)
    }

    func set(_ data: Data, for key: String) throws {
        // 先删除已有项
        try? delete(key)

        var query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
            kSecValueData as String: data,
            kSecAttrAccessible as String: kSecAttrAccessibleWhenUnlockedThisDeviceOnly,
        ]

        if let accessGroup = accessGroup {
            query[kSecAttrAccessGroup as String] = accessGroup
        }

        let status = SecItemAdd(query as CFDictionary, nil)
        guard status == errSecSuccess else {
            throw KeychainError.saveFailed(status)
        }
    }

    func get(_ key: String) throws -> String? {
        guard let data = try getData(key) else { return nil }
        return String(data: data, encoding: .utf8)
    }

    func getData(_ key: String) throws -> Data? {
        var query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne,
        ]

        if let accessGroup = accessGroup {
            query[kSecAttrAccessGroup as String] = accessGroup
        }

        var result: AnyObject?
        let status = SecItemCopyMatching(query as CFDictionary, &result)

        guard status != errSecItemNotFound else { return nil }
        guard status == errSecSuccess else {
            throw KeychainError.readFailed(status)
        }

        return result as? Data
    }

    func delete(_ key: String) throws {
        var query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: key,
        ]

        if let accessGroup = accessGroup {
            query[kSecAttrAccessGroup as String] = accessGroup
        }

        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw KeychainError.deleteFailed(status)
        }
    }

    func clear() throws {
        var query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
        ]

        if let accessGroup = accessGroup {
            query[kSecAttrAccessGroup as String] = accessGroup
        }

        let status = SecItemDelete(query as CFDictionary)
        guard status == errSecSuccess || status == errSecItemNotFound else {
            throw KeychainError.deleteFailed(status)
        }
    }
}

// MARK: - Keychain Error

enum KeychainError: LocalizedError {
    case encodingFailed
    case saveFailed(OSStatus)
    case readFailed(OSStatus)
    case deleteFailed(OSStatus)

    var errorDescription: String? {
        switch self {
        case .encodingFailed: return "数据编码失败"
        case .saveFailed(let status): return "Keychain 保存失败: \(status)"
        case .readFailed(let status): return "Keychain 读取失败: \(status)"
        case .deleteFailed(let status): return "Keychain 删除失败: \(status)"
        }
    }
}
```

```swift
protocol TokenProviderProtocol: AnyObject {
    func getAccessToken() async -> String?
    func getRefreshToken() async -> String?
    func saveTokens(access: String, refresh: String) async throws
    func refreshToken() async throws
    func clearTokens() async

/// Token 提供者实现
final class TokenProvider: TokenProviderProtocol {
    private let keychain: KeychainManagerProtocol

    private enum Keys {
        static let accessToken = "com.app.token.access"
        static let refreshToken = "com.app.token.refresh"
    }

    init(keychain: KeychainManagerProtocol) {
        self.keychain = keychain
    }

    func getAccessToken() async -> String? {
        try? keychain.get(Keys.accessToken)
    }

    func getRefreshToken() async -> String? {
        try? keychain.get(Keys.refreshToken)
    }

    func saveTokens(access: String, refresh: String) async throws {
        try keychain.set(access, for: Keys.accessToken)
        try keychain.set(refresh, for: Keys.refreshToken)
    }

    func refreshToken() async throws {
        guard let refreshToken = await getRefreshToken() else {
            throw AppError.authentication
        }

        // 调用刷新 Token API
        let response: RefreshTokenResponse = try await APIClient.shared.request(
            AuthAPI.refreshToken(refreshToken)
        )

        try await saveTokens(access: response.accessToken, refresh: response.refreshToken)
    }

    func clearTokens() async {
        try? keychain.delete(Keys.accessToken)
        try? keychain.delete(Keys.refreshToken)
    }
}
```

```swift
    func get<T>(_ key: UserDefaultsKey) -> T?
    func remove(_ key: UserDefaultsKey)
    func clear()
}

/// UserDefaults Key 定义
enum UserDefaultsKey: String, CaseIterable {
    case isFirstLaunch
    case isLoggedIn
    case userId
    case lastSyncDate
    case appTheme
    case language
    case fontSize
}

/// 用户偏好管理器
final class UserDefaultsManager: UserDefaultsManagerProtocol {
    private let defaults: UserDefaults
    private let suiteName: String?

    init(suiteName: String? = nil) {
        self.suiteName = suiteName
        if let suiteName = suiteName {
            defaults = UserDefaults(suiteName: suiteName) ?? .standard
        } else {
            defaults = .standard
        }
    }

    func set<T>(_ value: T, for key: UserDefaultsKey) {
        defaults.set(value, forKey: key.rawValue)
        defaults.synchronize()
    }

    func get<T>(_ key: UserDefaultsKey) -> T? {
        defaults.object(forKey: key.rawValue) as? T
    }

    func remove(_ key: UserDefaultsKey) {
        defaults.removeObject(forKey: key.rawValue)
    }

    func clear() {
        UserDefaultsKey.allCases.forEach { remove($0) }
    }
}

// 使用 @AppStorage（SwiftUI）
extension UserDefaultsKey {
    var appStorage: String { rawValue }
}

// 使用示例
struct SettingsView: View {
```

## 4. 缓存系统

### 4.1 多级缓存

```swift
// Core/Storage/CacheManager.swift

import Foundation

/// 缓存管理器协议
protocol CacheManagerProtocol {
    /// 获取缓存对象
    func object<T: Codable>(for key: String) async throws -> T?
    /// 设置缓存
    func set<T: Codable>(_ object: T, for key: String, expiry: CacheExpiry) async throws
    /// 删除缓存
    func remove(for key: String) async
    /// 清空所有缓存
    func clear() async
    /// 清空过期缓存
    func clearExpired() async
}

/// 缓存过期时间
enum CacheExpiry {
    case never
    case seconds(TimeInterval)
    case minutes(TimeInterval)
    case hours(TimeInterval)
    case days(TimeInterval)

    var timeInterval: TimeInterval {
        switch self {
        case .never: return .infinity
        case .seconds(let s): return s
        case .minutes(let m): return m * 60
        case .hours(let h): return h * 3600
        case .days(let d): return d * 86400
        }
    }
}

/// 缓存条目
struct CacheEntry: Codable {
    let data: Data
    let expiryDate: Date
    let createdAt: Date

    var isExpired: Bool {
        Date() > expiryDate
    }

    init<T: Codable>(value: T, expiry: CacheExpiry) throws {
        self.data = try JSONEncoder().encode(value)
        self.expiryDate = Date().addingTimeInterval(expiry.timeInterval)
        self.createdAt = Date()
    }

    func decode<T: Codable>() throws -> T {
        try JSONDecoder().decode(T.self, from: data)
    }
}

/// 缓存管理器实现（内存 + 磁盘二级缓存）
final class CacheManager: CacheManagerProtocol {
    private let memoryCache = NSCache<NSString, CacheEntry>()
    private let fileManager = FileManager.default
    private let cacheDirectory: URL
    private let decoder = JSONDecoder()
    private let encoder = JSONEncoder()
    private let queue = DispatchQueue(label: "com.app.cache", qos: .utility)

    init(cacheName: String = "default") {
        cacheDirectory = fileManager.urls(for: .cachesDirectory, in: .userDomainMask)
            .first!
            .appendingPathComponent("com.app.cache.\(cacheName)", isDirectory: true)

        try? fileManager.createDirectory(at: cacheDirectory,
                                         withIntermediateDirectories: true)

        // 内存缓存配置
        memoryCache.countLimit = 200
        memoryCache.totalCostLimit = 50 * 1024 * 1024 // 50MB

        // 启动时清理过期缓存
        Task { await clearExpired() }
    }

    func object<T: Codable>(for key: String) async throws -> T? {
        let sanitizedKey = sanitize(key)

        // 1. 检查内存缓存
        if let entry = memoryCache.object(forKey: sanitizedKey as NSString) {
            if entry.isExpired {
                memoryCache.removeObject(forKey: sanitizedKey as NSString)
            } else {
                return try entry.decode()
            }
        }

        // 2. 检查磁盘缓存
        return try await queue.sync {
            let fileURL = cacheDirectory.appendingPathComponent(sanitizedKey)
            guard fileManager.fileExists(atPath: fileURL.path) else { return nil as T? }

            let data = try Data(contentsOf: fileURL)
            let entry = try decoder.decode(CacheEntry.self, from: data)

            if entry.isExpired {
                try fileManager.removeItem(at: fileURL)
                return nil as T?
            }

            // 写回内存缓存
            memoryCache.setObject(entry, forKey: sanitizedKey as NSString)

            return try entry.decode()
        }
    }

    func set<T: Codable>(_ object: T, for key: String, expiry: CacheExpiry) async throws {
        let sanitizedKey = sanitize(key)
        let entry = try CacheEntry(value: object, expiry: expiry)

        // 写入内存缓存
        memoryCache.setObject(entry, forKey: sanitizedKey as NSString)

        // 写入磁盘缓存
        try await queue.sync {
            let fileURL = cacheDirectory.appendingPathComponent(sanitizedKey)
            let data = try encoder.encode(entry)
            try data.write(to: fileURL, options: .atomic)
        }
    }

    func remove(for key: String) async {
        let sanitizedKey = sanitize(key)
        memoryCache.removeObject(forKey: sanitizedKey as NSString)

        await queue.sync {
            let fileURL = cacheDirectory.appendingPathComponent(sanitizedKey)
            try? fileManager.removeItem(at: fileURL)
        }
    }

    func clear() async {
        memoryCache.removeAllObjects()

        await queue.sync {
            try? fileManager.removeItem(at: cacheDirectory)
            try? fileManager.createDirectory(at: cacheDirectory,
                                             withIntermediateDirectories: true)
        }
    }

    func clearExpired() async {
        await queue.sync {
            guard let enumerator = fileManager.enumerator(
                at: cacheDirectory,
                includingPropertiesForKeys: [.contentModificationDateKey]
            ) else { return }

            for case let fileURL as URL in enumerator {
                guard let data = try? Data(contentsOf: fileURL),
                      let entry = try? decoder.decode(CacheEntry.self, from: data),
                      entry.isExpired else { continue }
                try? fileManager.removeItem(at: fileURL)
            }
        }
    }

    private func sanitize(_ key: String) -> String {
        key.data(using: .utf8)?
            .base64EncodedString()
            .replacingOccurrences(of: "/", with: "_")
            .replacingOccurrences(of: "=", with: "") ?? key
    }
}
```

### 7.2 缓存策略模式

```swift
/// 缓存策略
enum CacheStrategy {
    /// 只读缓存，不请求网络
    case readOnly
    /// 先读缓存，再请求网络（缓存优先）
    case cacheFirst(expiry: CacheExpiry)
    /// 先请求网络，再更新缓存（网络优先）
    case networkFirst(expiry: CacheExpiry)
    /// 只请求网络，不读写缓存
    case networkOnly
    /// 缓存刷新：有缓存返回缓存，同时后台刷新
    case refreshStale(expiry: CacheExpiry)
}

/// 带缓存策略的 Repository 基类
protocol CacheableRepository {
    associatedtype T: Codable
    var cacheManager: CacheManagerProtocol { get }
    var cacheKey: String { get }

    func fetch(with strategy: CacheStrategy) async throws -> T
}

extension CacheableRepository {
    func fetch(with strategy: CacheStrategy) async throws -> T {
        switch strategy {
        case .readOnly:
            guard let cached: T = try await cacheManager.object(for: cacheKey) else {
                throw AppError.data(CacheError.notFound)
            }
            return cached

        case .cacheFirst(let expiry):
            if let cached: T = try await cacheManager.object(for: cacheKey) {
                return cached
            }
            let fresh = try await fetchFromNetwork()
            try await cacheManager.set(fresh, for: cacheKey, expiry: expiry)
            return fresh

        case .networkFirst(let expiry):
            let fresh = try await fetchFromNetwork()
            try await cacheManager.set(fresh, for: cacheKey, expiry: expiry)
            return fresh

        case .networkOnly:
            return try await fetchFromNetwork()

        case .refreshStale(let expiry):
            if let cached: T = try await cacheManager.object(for: cacheKey) {
                // 后台刷新
                Task {
                    if let fresh = try? await fetchFromNetwork() {
                        try? await cacheManager.set(fresh, for: cacheKey, expiry: expiry)
                    }
                }
                return cached
            }
            let fresh = try await fetchFromNetwork()
            try await cacheManager.set(fresh, for: cacheKey, expiry: expiry)
            return fresh
        }
```
