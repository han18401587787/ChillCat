## 1. 错误处理架构

### 1.1 错误层级

```
┌────────────────────────────────────────────┐
│              Presentation                   │
│   AppError → 用户友好的错误提示               │
│   (展示层错误处理)                           │
├────────────────────────────────────────────┤
│               Domain                        │
│   AppError → 业务错误码 + 消息               │
│   (业务层错误处理)                           │
├────────────────────────────────────────────┤
│                Data                         │
│   APIError → 网络错误 / 数据解析错误          │
│   (数据层错误处理)                           │
├────────────────────────────────────────────┤
│                Core                         │
│   Foundation Error → 系统级错误              │
│   (基础层错误)                               │
└────────────────────────────────────────────┘
```

### 1.2 错误转换流程

```
系统错误 / APIError / 其他 Error
    │
    ▼
catch 并转换为 AppError
    │
    ▼
AppError 传递到 ViewModel
    │
    ▼
ViewModel 将 AppError 映射为用户可读消息
    │
    ▼
View 显示错误提示
```

## 2. 统一错误模型

### 2.1 AppError 定义

```swift
// Core/Helpers/AppError.swift

import Foundation

/// 应用级统一错误
enum AppError: LocalizedError {
    // MARK: - 网络错误
    case network(APIError)
    case timeout
    case cancelled

    // MARK: - 业务错误
    case business(code: Int, message: String)
    case validation(ValidationError)

    // MARK: - 数据错误
    case data(Error)
    case notFound
    case decodingFailed

    // MARK: - 认证错误
    case authentication
    case tokenExpired
    case permissionDenied

    // MARK: - 文件错误
    case fileNotFound(String)
    case fileWriteFailed(String)
    case fileReadFailed(String)

    // MARK: - 未知错误
    case unknown(Error?)
    case notImplemented

    // MARK: - LocalizedError

    var errorDescription: String? {
        switch self {
        case .network(let apiError):
            return apiError.errorDescription
        case .timeout:
            return "请求超时，请检查网络后重试"
        case .cancelled:
            return "请求已取消"
        case .business(_, let message):
            return message
        case .validation(let error):
            return error.errorDescription
        case .data(let error):
            return "数据处理失败: \(error.localizedDescription)"
        case .notFound:
            return "数据不存在"
        case .decodingFailed:
            return "数据解析失败"
        case .authentication, .tokenExpired:
            return "登录已过期，请重新登录"
        case .permissionDenied:
            return "没有权限执行此操作"
        case .fileNotFound(let path):
            return "文件未找到: \(path)"
        case .fileWriteFailed(let path):
            return "文件写入失败: \(path)"
        case .fileReadFailed(let path):
            return "文件读取失败: \(path)"
        case .unknown(let error):
            return error?.localizedDescription ?? "未知错误"
        case .notImplemented:
            return "功能尚未实现"
        }
    }

    /// 错误码
    var code: Int {
        switch self {
        case .network(let apiError):
            switch apiError {
            case .unauthorized: return 401
            case .serverError(let c): return c
            default: return -1
            }
        case .timeout: return -1001
        case .business(let code, _): return code
        case .authentication, .tokenExpired: return 401
        case .validation: return 400
        case .notFound: return 404
        default: return -9999
        }
    }

    /// 是否可重试
    var isRetryable: Bool {
        switch self {
        case .network, .timeout, .business, .unknown, .serverError:
            return true
        case .cancelled, .authentication, .tokenExpired, .permissionDenied:
            return false
        case .validation, .data, .notFound, .decodingFailed, .fileNotFound, .fileWriteFailed, .fileReadFailed:
            return false
        }
    }
}

// MARK: - ValidationError

enum ValidationError: LocalizedError {
    case emptyUsername
    case emptyPassword
    case invalidEmail
    case passwordTooShort(min: Int)
    case invalidPhoneNumber
    case invalidURL
    case custom(String)

    var errorDescription: String? {
        switch self {
        case .emptyUsername: return "请输入用户名"
        case .emptyPassword: return "请输入密码"
        case .invalidEmail: return "邮箱格式不正确"
        case .passwordTooShort(let min): return "密码长度不能少于 \(min) 位"
        case .invalidPhoneNumber: return "手机号格式不正确"
        case .invalidURL: return "URL 格式不正确"
        case .custom(let message): return message
        }
    }
}

// MARK: - Error 转换

extension Error {
    /// 转换为 AppError
    var asAppError: AppError {
        switch self {
        case let appError as AppError:
            return appError
        case let apiError as APIError:
            return .network(apiError)
        case let validationError as ValidationError:
            return .validation(validationError)
        case let urlError as URLError:
            switch urlError.code {
            case .timedOut: return .timeout
            case .cancelled: return .cancelled
            case .notConnectedToInternet: return .network(.networkFailure(urlError))
            case .networkConnectionLost:  return .network(.networkFailure(urlError))
            default: return .network(.networkFailure(urlError))
            }
        case let decodingError as DecodingError:
            return .decodingFailed(decodingError)
        case let nsError as NSError where nsError.domain == "NSCocoaErrorDomain":
            switch nsError.code {
            case NSFileReadNoSuchFileError, NSFileNoSuchFileError:
                return .fileNotFound("")
            case NSFileWriteOutOfSpaceError:
                return .fileWriteFailed("存储空间不足")
            default:
                return .unknown(self)
            }
        default:
            return .unknown(self)
        }
    }
}
```

## 3. 错误处理实践

### 3.1 UseCase 层错误处理

```swift
// Domain/UseCases/LoginUseCase.swift

final class LoginUseCase: LoginUseCaseProtocol {
    private let userRepository: UserRepositoryProtocol

    init(userRepository: UserRepositoryProtocol) {
        self.userRepository = userRepository
    }

    func execute(username: String, password: String) async throws -> User {
        // 参数校验
        guard !username.trimmingCharacters(in: .whitespaces).isEmpty else {
            throw AppError.validation(.emptyUsername)
        }
        guard !password.isEmpty else {
            throw AppError.validation(.emptyPassword)
        }
        guard password.count >= 6 else {
            throw AppError.validation(.passwordTooShort(min: 6))
        }

        // 业务逻辑
        do {
            return try await userRepository.login(username: username, password: password)
        } catch {
            // 统一转换为 AppError
            throw error.asAppError
        }
    }
}
```

### 3.2 ViewModel 层错误处理

```swift
// Presentation/Modules/Auth/ViewModels/LoginViewModel.swift

@MainActor
@Observable
final class LoginViewModel {
    var state = LoginState()

    private let loginUseCase: LoginUseCaseProtocol

    init(loginUseCase: LoginUseCaseProtocol) {
        self.loginUseCase = loginUseCase
    }

    func login(username: String, password: String) async {
        state.isLoading = true
        state.error = nil

        do {
            let user = try await loginUseCase.execute(
                username: username,
                password: password
            )
            state.isLoggedIn = true
            Logger.info("用户登录成功: \(user.id)")
        } catch {
            let appError = error.asAppError
            state.error = appError

            // 认证错误特殊处理
            if case .authentication = appError {
                state.shouldRedirectToLogin = true
            }

            Logger.error("登录失败: \(appError.errorDescription ?? "")")
        }

        state.isLoading = false
    }
}
```

## 4. 日志系统

### 4.1 全链路追踪（Distributed Tracing）

#### 4.1.1 TraceContext — 追踪上下文

```swift
import Foundation

/// 全链路追踪上下文，贯穿整个请求/操作生命周期
public struct TraceContext: Sendable, Codable {
    /// 全局唯一追踪 ID，一次用户操作从头到尾共享
    public let traceID: String
    /// 当前 Span ID，标识当前操作单元
    public let spanID: String
    /// 父 Span ID，用于构建调用链
    public let parentSpanID: String?
    /// 操作名称
    public let operationName: String
    /// 开始时间
    public let startTime: Date
    /// 自定义标签
    public var tags: [String: String]

    public init(
        traceID: String = UUID().uuidString.replacingOccurrences(of: "-", with: "").lowercased(),
        spanID: String = UUID().uuidString.replacingOccurrences(of: "-", with: "").lowercased().prefix(16).lowercased(),
        parentSpanID: String? = nil,
        operationName: String,
        startTime: Date = Date(),
        tags: [String: String] = [:]
    ) {
        self.traceID = traceID
        self.spanID = spanID
        self.parentSpanID = parentSpanID
        self.operationName = operationName
        self.startTime = startTime
        self.tags = tags
    }

    /// 创建子 Span，继承当前 TraceID
    public func makeChild(operationName: String) -> TraceContext {
        TraceContext(
            traceID: traceID,
            parentSpanID: spanID,
            operationName: operationName,
            tags: tags
        )
    }
}
```

#### 4.1.2 TraceManager — 追踪管理器

```swift
import Foundation

/// 全链路追踪管理器
@MainActor
@Observable
public final class TraceManager {
    public static let shared = TraceManager()
    /// 当前活跃的 Trace 栈
    private var traceStack: [TraceContext] = []
    /// 已完成的 Span 记录（用于导出）
    private var completedSpans: [TraceContext] = []
    /// 最大保留 Span 数
    private let maxSpans: Int = 1000

    private init() {}

    /// 开始一个新的根 Trace
    @discardableResult
    public func startTrace(operationName: String) -> TraceContext {
        let trace = TraceContext(operationName: operationName)
        traceStack = [trace]
        return trace
    }

    /// 开始一个子 Span
    @discardableResult
    public func startSpan(operationName: String) -> TraceContext {
        guard let parent = traceStack.last else {
            return startTrace(operationName: operationName)
        }
        let child = parent.makeChild(operationName: operationName)
        traceStack.append(child)
        return child
    }

    /// 结束当前 Span
    public func endSpan() {
        guard !traceStack.isEmpty else { return }
        let completed = traceStack.removeLast()
        completedSpans.append(completed)
        if completedSpans.count > maxSpans {
            completedSpans.removeFirst(completedSpans.count - maxSpans)
        }
    }

    /// 获取当前 TraceID
    public var currentTraceID: String? { traceStack.last?.traceID }
    /// 获取当前 Span
    public var currentSpan: TraceContext? { traceStack.last }
    /// 导出所有完成的 Span（用于上报）
    public func exportSpans() -> [TraceContext] {
        let spans = completedSpans
        completedSpans.removeAll()
        return spans
    }
}
```

#### 4.1.3 TraceInterceptor — 网络层追踪中间件

```swift
import Foundation

/// 网络层追踪中间件，自动为每个请求注入 TraceID
public final class TraceInterceptor: RequestInterceptor {
    private let traceManager: TraceManager
    private let logger: LoggerProtocol

    public init(traceManager: TraceManager = .shared,
                logger: LoggerProtocol = Logger.shared) {
        self.traceManager = traceManager
        self.logger = logger
    }

    public func intercept(_ request: URLRequest) async throws -> URLRequest {
        var modifiedRequest = request
        if let traceID = traceManager.currentTraceID {
            modifiedRequest.setValue(traceID, forHTTPHeaderField: "X-Trace-ID")
        }
        if let spanID = traceManager.currentSpan?.spanID {
            modifiedRequest.setValue(spanID, forHTTPHeaderField: "X-Span-ID")
        }
        return modifiedRequest
    }

    public func handle(response: HTTPURLResponse, data: Data) async throws {
        if let serverTraceID = response.allHeaderFields["X-Trace-ID"] as? String {
            logger.debug("服务端 TraceID: " + serverTraceID)
        }
    }
}
```

#### 4.2.3 LogEntry — 结构化日志条目

```swift
import Foundation

/// 结构化日志条目，统一 JSON 格式
public struct LogEntry: Sendable, Codable {
    public let timestamp: Date
    public let level: String
    public let module: String
    public let category: String
    public let message: String
    public let traceID: String?
    public let spanID: String?
    public let file: String
    public let function: String
    public let line: Int
    public let tags: [String: String]?
    public let metadata: [String: String]?
    public let error: LogErrorInfo?

    public struct LogErrorInfo: Sendable, Codable {
        public let domain: String
        public let code: Int
        public let description: String
        public let stackTrace: String?
    }

    /// 序列化为 JSON 字符串
    public func toJSONString() -> String? {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.sortedKeys]
        encoder.dateEncodingStrategy = .iso8601
        return (try? encoder.encode(self)).flatMap { String(data: $0, encoding: .utf8) }
    }

    /// 格式化为可读文本
    public func toReadableString() -> String {
        let dateStr = ISO8601DateFormatter().string(from: timestamp)
        var result = dateStr + " [" + level + "] [" + module + "] " + message
        if let traceID = traceID { result += " | trace:" + traceID }
        if let fileInfo = file.split(separator: "/").last { result += " (" + fileInfo + ":" + String(line) + ")" }
        return result
    }
}
```

### 4.4 日志实现

```swift
/// 日志实现（基于 OSLog，支持模块化 + 文件写入 + 远程上报）
final class Logger: LoggerProtocol {
    static let shared = Logger()

    private let osLog: OSLog
    private let queue: DispatchQueue
    private let traceManager: TraceManager
    private var config: LogConfig
    private var moduleLevelOverrides: [LogModule: LogLevel]

    init(subsystem: String = Bundle.main.bundleIdentifier ?? "com.app",
         traceManager: TraceManager = .shared) {
        self.osLog = OSLog(subsystem: subsystem, category: "App")
        self.queue = DispatchQueue(label: "com.app.logger", qos: .utility)
        self.traceManager = traceManager
        self.config = .default(environment: .development)
        self.moduleLevelOverrides = [:]
    }

    func configure(_ config: LogConfig) {
        queue.sync { self.config = config }
    }

    func level(for module: LogModule) -> LogLevel {
        queue.sync { moduleLevelOverrides[module] ?? config.moduleLevels[module] ?? config.defaultLevel }
    }

    func setLevel(_ level: LogLevel, for module: LogModule) {
        queue.sync { moduleLevelOverrides[module] = level }
    }

    func debug(_ message: String, module: LogModule = .default, category: String = "General",
               file: String = #file, function: String = #function, line: Int = #line) {
        log(level: .debug, message: message, module: module, category: category,
            file: file, function: function, line: line, error: nil)
    }

    func info(_ message: String, module: LogModule = .default, category: String = "General",
              file: String = #file, function: String = #function, line: Int = #line) {
        log(level: .info, message: message, module: module, category: category,
            file: file, function: function, line: line, error: nil)
    }

    func warning(_ message: String, module: LogModule = .default, category: String = "General",
                 file: String = #file, function: String = #function, line: Int = #line) {
        log(level: .warning, message: message, module: module, category: category,
            file: file, function: function, line: line, error: nil)
    }

    func error(_ message: String, module: LogModule = .default, category: String = "General",
               file: String = #file, function: String = #function, line: Int = #line, error: Error? = nil) {
        log(level: .error, message: message, module: module, category: category,
            file: file, function: function, line: line, error: error)
    }

    private func log(level: LogLevel, message: String, module: LogModule, category: String,
                     file: String, function: String, line: Int, error: Error?) {
        let minLevel = moduleLevelOverrides[module] ?? config.moduleLevels[module] ?? config.defaultLevel
        guard level >= minLevel else { return }

        let fileName = (file as NSString).lastPathComponent
        let traceID = config.includeTrace ? traceManager.currentTraceID : nil
        let spanID = config.includeTrace ? traceManager.currentSpan?.spanID : nil

        var errorInfo: LogEntry.LogErrorInfo? = nil
        if let err = error {
            let appError = err.asAppError
            errorInfo = LogEntry.LogErrorInfo(
                domain: appError.domain, code: appError.errorCode,
                description: appError.errorDescription ?? err.localizedDescription,
                stackTrace: config.includeCallStack ? Thread.callStackSymbols.joined(separator: "\n") : nil
            )
        }

        let entry = LogEntry(
            timestamp: Date(), level: level.label, module: module.rawValue,
            category: category, message: message, traceID: traceID, spanID: spanID,
            file: fileName, function: function, line: line,
            tags: traceManager.currentSpan?.tags, metadata: nil, error: errorInfo
        )

        if config.consoleOutput {
            os_log("%{public}@", log: osLog, type: level.osLogType, entry.toReadableString())
        }
        if config.fileOutput { writeToFile(entry) }
        if config.remoteReporting && level >= .error { reportToRemote(entry) }
    }

    private func writeToFile(_ entry: LogEntry) {
        queue.async { [weak self] in
            guard let self = self, let dir = self.config.logDirectory ?? self.defaultLogDirectory() else { return }
            let dateStr = ISO8601DateFormatter().string(from: Date()).replacingOccurrences(of: ":", with: "-")
            let fileURL = URL(fileURLWithPath: dir).appendingPathComponent("app-\(dateStr).log")
            guard let jsonString = entry.toJSONString(), let data = (jsonString + "\n").data(using: .utf8) else { return }

            if FileManager.default.fileExists(atPath: fileURL.path) {
                if let handle = try? FileHandle(forWritingTo: fileURL) {
                    handle.seekToEndOfFile(); handle.write(data); try? handle.close()
                }
            } else {
                try? data.write(to: fileURL, options: .atomic)
            }
        }
    }

    private func reportToRemote(_ entry: LogEntry) {
        Task { [entry] in await ErrorReporter.shared.reportLogEntry(entry) }
    }

    private func defaultLogDirectory() -> String? {
        guard let cacheDir = NSSearchPathForDirectoriesInDomains(.cachesDirectory, .userDomainMask, true).first else { return nil }
        return (cacheDir as NSString).appendingPathComponent("Logs")
    }
}
```

```swift
// Presentation/Common/Components/ToastView.swift

import SwiftUI

/// Toast 类型
enum ToastType {
    case success
    case error
    case warning
    case info

    var icon: String {
        switch self {
        case .success: return "checkmark.circle.fill"
        case .error: return "xmark.circle.fill"
        case .warning: return "exclamationmark.triangle.fill"
        case .info: return "info.circle.fill"
        }
    }

    var color: Color {
        switch self {
        case .success: return .green
        case .error: return .red
        case .warning: return .orange
        case .info: return .blue
        }
    }
}

/// Toast 视图
struct ToastView: View {
    let message: String
    let type: ToastType

    var body: some View {
        HStack(spacing: 8) {
            Image(systemName: type.icon)
                .foregroundColor(type.color)
            Text(message)
                .font(.subheadline)
                .foregroundColor(.primary)
        }
        .padding(.horizontal, 16)
        .padding(.vertical, 12)
        .background(.regularMaterial)
        .cornerRadius(8)
        .shadow(radius: 4)
        .padding(.horizontal)
    }
}

/// Toast 修饰器
struct ToastModifier: ViewModifier {
    @Binding var message: String?
    @State private var show = false

    func body(content: Content) -> some View {
        content
            .overlay(alignment: .top) {
                if show, let message = message {
                    ToastView(message: message, type: .info)
                        .transition(.move(edge: .top).combined(with: .opacity))
                        .onAppear {
                            DispatchQueue.main.asyncAfter(deadline: .now() + 2) {
                                withAnimation {
                                    self.message = nil
                                    show = false
                                }
                            }
                        }
                }
            }
            .onChange(of: message) { _, newValue in
                withAnimation {
                    show = newValue != nil
                }
            }
    }
}

extension View {
    func toast(message: Binding<String?>) -> some View {
        modifier(ToastModifier(message: message))
    }
}
```

```swift
        switch self {
        case .authentication, .tokenExpired:
            return .alert
        case .validation:
            return .inline
        case .network, .timeout:
            return .toast
        case .notFound:
```

## 6. 错误监控

### 6.1 错误上报

```swift
// Core/Logging/ErrorReporter.swift

import Foundation

/// 错误上报协议
protocol ErrorReporterProtocol {
    func report(_ error: AppError, context: [String: Any]?)
    func report(_ error: Error, context: [String: Any]?)
    func reportMessage(_ message: String, level: LogLevel, context: [String: Any]?)
    func reportLogEntry(_ entry: LogEntry)
    func reportEvent(_ name: String, parameters: [String: Any]?)
}

// MARK: - ErrorReporterProtocol 默认实现
extension ErrorReporterProtocol {
    func reportLogEntry(_ entry: LogEntry) {}
    func reportEvent(_ name: String, parameters: [String: Any]?) {}
}

/// 错误上报器，聚合多个上报通道，支持采样率控制
final class ErrorReporter: ErrorReporterProtocol {
    static let shared = ErrorReporter()

    private var reporters: [ErrorReporterProtocol] = []
    private var isEnabled: Bool = true
    private var sampleRate: Double = 1.0

    private init() {}

    func addReporter(_ reporter: ErrorReporterProtocol) { reporters.append(reporter) }
    func setSampleRate(_ rate: Double) { sampleRate = max(0.0, min(1.0, rate)) }
    func setEnabled(_ enabled: Bool) { isEnabled = enabled }

    func report(_ error: AppError, context: [String: Any]? = nil) {
        guard isEnabled, shouldSample() else { return }
        var enrichedContext = context ?? [:]
        enrichedContext["traceID"] = TraceManager.shared.currentTraceID
        enrichedContext["domain"] = error.domain
        enrichedContext["errorCode"] = error.errorCode
        enrichedContext["isRetryable"] = error.isRetryable
        reporters.forEach { $0.report(error, context: enrichedContext) }
    }

    func report(_ error: Error, context: [String: Any]? = nil) {
        report(error.asAppError, context: context)
    }

    func reportMessage(_ message: String, level: LogLevel, context: [String: Any]? = nil) {
        Logger.info("非崩溃日志上报: \(message)")
    }

    func reportLogEntry(_ entry: LogEntry) {
        guard isEnabled, shouldSample() else { return }
        reporters.forEach { $0.reportLogEntry(entry) }
    }

    func reportEvent(_ name: String, parameters: [String: Any]? = nil) {
        guard isEnabled, shouldSample() else { return }
        reporters.forEach { $0.reportEvent(name, parameters: parameters) }
    }

    private func shouldSample() -> Bool {
        guard sampleRate < 1.0 else { return true }
        return Double.random(in: 0...1) < sampleRate
    }
}
```

```swift
         logger: LoggerProtocol = Logger.shared) {
        self.loginUseCase = loginUseCase

            logger.error("登录失败", module: .auth, category: "Login", error: error)

        }
    }
```
