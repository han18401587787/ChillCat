## 2. 可扩展性策略

### 2.1 扩展点设计

| 扩展场景 | 设计模式 | 实现方式 |

```swift
// Core/Helpers/FeatureFlag.swift

import Foundation

/// 功能开关
enum FeatureFlag: String, CaseIterable {
    case newHomePage = "new_home_page"
    case darkMode = "dark_mode"
    case experimentalPlayer = "experimental_player"
    case newLoginFlow = "new_login_flow"
    case aiRecommendation = "ai_recommendation"

    /// 是否启用
    var isEnabled: Bool {
        // 优先使用远程配置
        if let remoteValue = RemoteConfig.shared.bool(forKey: rawValue) {
            return remoteValue
        }
        // 回退到本地默认值
        return defaultEnabled
    }

    /// 本地默认值
    var defaultEnabled: Bool {
        switch self {
        case .newHomePage: return false
        case .darkMode: return true
        case .experimentalPlayer: return false
        case .newLoginFlow: return false
        case .aiRecommendation: return false
        }
    }

    /// 强制覆盖（用于开发调试）
    var isOverridden: Bool {
        UserDefaults.standard.object(forKey: "ff_override_\(rawValue)") != nil
    }

    var overriddenValue: Bool {
        UserDefaults.standard.bool(forKey: "ff_override_\(rawValue)")
    }

    /// 在调试菜单中覆盖
    func override(_ value: Bool) {
        UserDefaults.standard.set(value, forKey: "ff_override_\(rawValue)")
    }

    func resetOverride() {
        UserDefaults.standard.removeObject(forKey: "ff_override_\(rawValue)")
    }
}
```

```swift
// ✅ 请求合并
final class RequestBatcher {
    private var queue: [() async -> Void] = []
    private let interval: TimeInterval

    init(interval: TimeInterval = 0.3) {
        self.interval = interval
    }

    func add(_ request: @escaping () async -> Void) {
        queue.append(request)
        // 防抖处理
        DispatchQueue.main.asyncAfter(deadline: .now() + interval) { [weak self] in
            self?.flush()
        }
    }

    private func flush() {
        let batch = queue
        queue = []
        Task {
            await withTaskGroup(of: Void.self) { group in
                for request in batch {
                    group.addTask { await request() }
                }
            }
        }
    }
}

// ✅ 数据预加载
func prefetchData() {
    Task {
        async let feeds = feedRepository.fetchFeeds()
        async let user = userRepository.fetchProfile()
        let (_, _) = try await (feeds, user)
    }
}
```

### 4.2 SSL Pinning

```swift
// Core/Networking/SSLPinningDelegate.swift

import Foundation

/// SSL Pinning 代理
final class SSLPinningDelegate: NSObject, URLSessionDelegate {
    private let pinnedCertificates: [SecCertificate]

    init(certificateNames: [String]) {
        self.pinnedCertificates = certificateNames.compactMap { name in
            guard let path = Bundle.main.path(forResource: name, ofType: "cer"),
                  let data = try? Data(contentsOf: URL(fileURLWithPath: path)),
                  let certificate = SecCertificateCreateWithData(nil, data as CFData) else {
                return nil
            }
            return certificate
        }
        super.init()
    }

    func urlSession(
        _ session: URLSession,
        didReceive challenge: URLAuthenticationChallenge
    ) async -> (URLSession.AuthChallengeDisposition, URLCredential?) {
        guard challenge.protectionSpace.authenticationMethod == NSURLAuthenticationMethodServerTrust,
              let serverTrust = challenge.protectionSpace.serverTrust else {
            return (.performDefaultHandling, nil)
        }

        // 获取服务器证书
        let serverCertCount = SecTrustGetCertificateCount(serverTrust)
        guard serverCertCount > 0,
              let serverCert = SecTrustGetCertificateAtIndex(serverTrust, 0) else {
            return (.cancelAuthenticationChallenge, nil)
        }

        // 验证证书
        let isPinned = pinnedCertificates.contains { pinnedCert in
            let serverCertData = SecCertificateCopyData(serverCert) as Data
            let pinnedCertData = SecCertificateCopyData(pinnedCert) as Data
            return serverCertData == pinnedCertData
        }

        if isPinned {
            return (.useCredential, URLCredential(trust: serverTrust))
        } else {
            return (.cancelAuthenticationChallenge, nil)
        }
    }
}
```

### 4.3 输入校验

```swift
// ✅ 所有用户输入必须校验
struct InputValidator {
    static func validateEmail(_ email: String) -> Bool {
        let emailRegex = "[A-Z0-9a-z._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,64}"
        return NSPredicate(format: "SELF MATCHES %@", emailRegex).evaluate(with: email)
    }

    static func validatePhone(_ phone: String) -> Bool {
        let phoneRegex = "^1[3-9]\\d{9}$"
        return NSPredicate(format: "SELF MATCHES %@", phoneRegex).evaluate(with: phone)
    }

    static func validatePassword(_ password: String) -> ValidationResult {
        if password.count < 6 {
            return .invalid("密码长度不能少于 6 位")
        }
        if !password.contains(where: { $0.isUppercase }) {
            return .invalid("密码需要包含大写字母")
        }
        if !password.contains(where: { $0.isNumber }) {
            return .invalid("密码需要包含数字")
        }
        return .valid
    }

    enum ValidationResult {
        case valid
        case invalid(String)
    }
}
```

### 5.1 性能监控

```swift
// Core/Monitoring/PerformanceMonitor.swift

import Foundation

/// 性能监控器
final class PerformanceMonitor {
    static let shared = PerformanceMonitor()

    private var metrics: [String: Date] = [:]

    private init() {}

    /// 开始计时
    func start(_ name: String) {
        metrics[name] = Date()
    }

    /// 结束计时并上报
    func end(_ name: String, threshold: TimeInterval = 2.0) {
        guard let startTime = metrics.removeValue(forKey: name) else { return }
        let duration = Date().timeIntervalSince(startTime)

        if duration > threshold {
            Logger.warning("性能告警: \(name) 耗时 \(duration)s (阈值: \(threshold)s)")
            // 上报到监控平台
        }

        Logger.debug("性能指标: \(name) = \(duration)s")
    }

    /// 测量异步操作耗时
    func measure<T>(_ name: String, threshold: TimeInterval = 2.0,
                    operation: () async throws -> T) async rethrows -> T {
        let start = Date()
        let result = try await operation()
        let duration = Date().timeIntervalSince(start)

        if duration > threshold {
            Logger.warning("性能告警: \(name) 耗时 \(duration)s")
        }

        return result
    }
}

// 使用示例
func loadData() async {
    PerformanceMonitor.shared.start("loadData")
    // ... 加载数据
    PerformanceMonitor.shared.end("loadData", threshold: 1.0)

    // 或使用 measure
    let result = try await PerformanceMonitor.shared.measure("fetchAPI") {
        try await apiClient.request(endpoint)
    }
}
```

### 5.2 崩溃监控

```swift
// Core/Monitoring/CrashReporter.swift

import Foundation

/// 崩溃报告器
final class CrashReporter {
    static let shared = CrashReporter()

    private init() {
        setupCrashHandlers()
    }

    private func setupCrashHandlers() {
        // 注册未捕获异常处理
        NSSetUncaughtExceptionHandler { exception in
            CrashReporter.shared.handleException(exception)
        }

        // 注册信号处理
        signal(SIGABRT) { _ in CrashReporter.shared.handleSignal("SIGABRT") }
        signal(SIGSEGV) { _ in CrashReporter.shared.handleSignal("SIGSEGV") }
        signal(SIGBUS)  { _ in CrashReporter.shared.handleSignal("SIGBUS") }
        signal(SIGTRAP) { _ in CrashReporter.shared.handleSignal("SIGTRAP") }
    }

    private func handleException(_ exception: NSException) {
        let info: [String: Any] = [
            "name": exception.name.rawValue,
            "reason": exception.reason ?? "",
            "callStack": exception.callStackSymbols
        ]
        Logger.error("未捕获异常: \(exception.reason ?? "")")
        // 上报到崩溃平台
    }

    private func handleSignal(_ signal: String) {
        Logger.error("信号崩溃: \(signal)")
        // 上报到崩溃平台
    }
}
```
