### 1.1 会员类型

|------|------|------|

### 3.1 会员类型

```swift
// Domain/Entities/CCMemberType.swift

/// 会员类型
enum CCMemberType: String, Codable, CaseIterable {
    case monthly    = "monthly"     // 月度会员
    case quarterly  = "quarterly"   // 季度会员
    case yearly     = "yearly"      // 年度会员
    case permanent  = "permanent"   // 永久会员

    var displayName: String {
        switch self {
        case .monthly:   return "月度会员"
        case .quarterly: return "季度会员"
        case .yearly:    return "年度会员"
        case .permanent: return "永久会员"
        }
    }

    var isSubscription: Bool {
        switch self {
        case .monthly, .quarterly, .yearly: return true
        case .permanent: return false
        }
    }
}
```

### 3.3 会员信息实体

```swift
// Domain/Entities/CCMemberInfo.swift

/// 会员信息
struct CCMemberInfo: Codable, Equatable {
    let type: CCMemberType          // 会员类型
    let status: CCMemberStatus      // 会员状态
    let startDate: Date             // 生效时间
    let endDate: Date?              // 到期时间（永久会员为 nil）
    let isAutoRenew: Bool           // 是否自动续费
    let purchaseDate: Date          // 购买时间

    /// 是否在有效期内
    var isValid: Bool {
        guard status == .active else { return false }
        guard let endDate else { return true } // 永久会员永久有效
        return endDate > Date()
    }

    /// 剩余天数（永久会员返回 nil）
    var remainingDays: Int? {
        guard let endDate else { return nil }
        return Calendar.current.dateComponents([.day], from: Date(), to: endDate).day
    }
}
```

## 4. Data 层设计

### 4.1 API 端点

```swift
// API/CCMemberAPI.swift

/// 会员模块 API 端点
enum CCMemberAPI {
    case info                    // 获取会员信息
    case products                // 获取商品列表
    case privileges              // 获取权益列表
    case purchase(productId: String)  // 购买确认
    case history                 // 购买历史
}

extension CCMemberAPI: APIEndpoint {
    var baseURL: URL { Environment.current.baseURL }

    var path: String {
        switch self {
        case .info:              return "/api/v1/member/info"
        case .products:          return "/api/v1/member/products"
        case .privileges:        return "/api/v1/member/privileges"
        case .purchase:          return "/api/v1/member/purchase"
        case .history:           return "/api/v1/member/history"
        }
    }

    var method: HTTPMethod {
        switch self {
        case .info, .products, .privileges, .history: return .get
        case .purchase: return .post
        }
    }

    var body: Encodable? {
        switch self {
        case .purchase(let productId):
            return ["product_id": productId]
        default:
            return nil
        }
    }
}
```

### 4.3 DTO Mapper

```swift
// DTOs/CCMemberDTOMapper.swift

/// DTO 转 Entity 映射器
enum CCMemberDTOMapper {
    static func toEntity(_ dto: CCMemberInfoDTO) -> CCMemberInfo {
        let formatter = ISO8601DateFormatter()
        return CCMemberInfo(
            type: CCMemberType(rawValue: dto.memberType) ?? .monthly,
            status: CCMemberStatus(rawValue: dto.status) ?? .none,
            startDate: formatter.date(from: dto.startDate) ?? Date(),
            endDate: dto.endDate.flatMap { formatter.date(from: $0) },
            isAutoRenew: dto.autoRenew,
            purchaseDate: formatter.date(from: dto.purchaseDate) ?? Date()
        )
    }

    static func toEntity(_ dto: CCMemberProductDTO) -> CCMemberProduct {
        CCMemberProduct(
            id: dto.productId,
            type: CCMemberType(rawValue: dto.memberType) ?? .monthly,
            price: Decimal(string: dto.price) ?? 0,
            originalPrice: dto.originalPrice.flatMap { Decimal(string: $0) },
            displayPrice: "¥\(dto.price)",
            discountTag: dto.discountTag
        )
    }

    static func toEntity(_ dto: CCMemberPrivilegeDTO) -> CCMemberPrivilege {
        CCMemberPrivilege(
            id: dto.privilegeId,
            title: dto.title,
            description: dto.description,
            iconName: dto.icon,
            isHighlight: dto.highlight,
            availableTypes: dto.availableTypes.compactMap { CCMemberType(rawValue: $0) }
        )
    }
}
```

### 4.4 RemoteDataSource

```swift
// DataSources/CCMemberRemoteDataSource.swift

/// 会员远程数据源
final class CCMemberRemoteDataSource {
    private let apiClient: APIClientProtocol

    init(apiClient: APIClientProtocol) {
        self.apiClient = apiClient
    }

    func fetchMemberInfo() async throws -> CCMemberInfoDTO {
        let response: CCMemberResponseDTO<CCMemberInfoDTO> = try await apiClient.request(CCMemberAPI.info)
        guard response.isSuccess, let data = response.data else {
            throw APIError.badRequest
        }
        return data
    }

    func fetchProducts() async throws -> [CCMemberProductDTO] {
        let response: CCMemberResponseDTO<[CCMemberProductDTO]> = try await apiClient.request(CCMemberAPI.products)
        return response.data ?? []
    }

    func fetchPrivileges() async throws -> [CCMemberPrivilegeDTO] {
        let response: CCMemberResponseDTO<[CCMemberPrivilegeDTO]> = try await apiClient.request(CCMemberAPI.privileges)
        return response.data ?? []
    }

    func purchase(productId: String) async throws -> CCMemberInfoDTO {
        let response: CCMemberResponseDTO<CCMemberInfoDTO> = try await apiClient.request(CCMemberAPI.purchase(productId: productId))
        guard response.isSuccess, let data = response.data else {
            throw APIError.badRequest
        }
        return data
    }
}
```

### 4.5 LocalDataSource

```swift
// DataSources/CCMemberLocalDataSource.swift

/// 会员本地数据源（缓存）
final class CCMemberLocalDataSource {
    private let userDefaults: UserDefaults
    private let memberInfoKey = "cc_cached_member_info"

    init(userDefaults: UserDefaults = .standard) {
        self.userDefaults = userDefaults
    }

    func cacheMemberInfo(_ info: CCMemberInfo) throws {
        let data = try JSONEncoder().encode(info)
        userDefaults.set(data, forKey: memberInfoKey)
    }

    func getCachedMemberInfo() throws -> CCMemberInfo? {
        guard let data = userDefaults.data(forKey: memberInfoKey) else { return nil }
        return try JSONDecoder().decode(CCMemberInfo.self, from: data)
    }

    func clearCache() {
        userDefaults.removeObject(forKey: memberInfoKey)
    }
}
```

### 4.6 Repository 实现

```swift
// Repositories/CCMemberRepository.swift

/// 会员仓库实现
final class CCMemberRepository: CCMemberRepositoryProtocol {
    private let remoteDataSource: CCMemberRemoteDataSource
    private let localDataSource: CCMemberLocalDataSource

    init(
        remoteDataSource: CCMemberRemoteDataSource,
        localDataSource: CCMemberLocalDataSource
    ) {
        self.remoteDataSource = remoteDataSource
        self.localDataSource = localDataSource
    }

    func fetchMemberInfo() async throws -> CCMemberInfo {
        // 先读缓存
        if let cached = try? localDataSource.getCachedMemberInfo() {
            return cached
        }
        // 请求网络
        let dto = try await remoteDataSource.fetchMemberInfo()
        let info = CCMemberDTOMapper.toEntity(dto)
        // 更新缓存
        try? localDataSource.cacheMemberInfo(info)
        return info
    }

    func fetchProducts() async throws -> [CCMemberProduct] {
        let dtos = try await remoteDataSource.fetchProducts()
        return dtos.map { CCMemberDTOMapper.toEntity($0) }
    }

    func fetchPrivileges() async throws -> [CCMemberPrivilege] {
        let dtos = try await remoteDataSource.fetchPrivileges()
        return dtos.map { CCMemberDTOMapper.toEntity($0) }
    }

    func purchase(product: CCMemberProduct) async throws -> CCMemberInfo {
        let dto = try await remoteDataSource.purchase(productId: product.id)
        let info = CCMemberDTOMapper.toEntity(dto)
        // 更新缓存
        try? localDataSource.cacheMemberInfo(info)
        return info
    }

    func restorePurchases() async throws -> CCMemberInfo? {
        // 恢复购买逻辑（依赖 StoreKit）
        // 恢复成功后重新拉取会员信息
        return try await fetchMemberInfo()
    }
}
```

## 5. Presentation 层设计

### 5.1 ViewModel

```swift
// ViewModels/CCMemberViewModel.swift

/// 会员中心 ViewModel
@MainActor
@Observable
final class CCMemberViewModel {
    // MARK: - State

    struct State {
        var memberInfo: CCMemberInfo?
        var products: [CCMemberProduct] = []
        var privileges: [CCMemberPrivilege] = []
        var isLoading = false
        var error: String?
    }

    var state = State()

    // MARK: - Dependencies

    private let fetchMemberInfoUseCase: CCFetchMemberInfoUseCase
    private let purchaseMemberUseCase: CCPurchaseMemberUseCase

    init(
        fetchMemberInfoUseCase: CCFetchMemberInfoUseCase,
        purchaseMemberUseCase: CCPurchaseMemberUseCase
    ) {
        self.fetchMemberInfoUseCase = fetchMemberInfoUseCase
        self.purchaseMemberUseCase = purchaseMemberUseCase
    }

    // MARK: - Actions

    func loadData() async {
        state.isLoading = true
        state.error = nil

        do {
            async let info = fetchMemberInfoUseCase.execute()
            async let products = CCMemberProduct.fetchDummyProducts() // 临时
            async let privileges = CCMemberPrivilege.fetchDummyPrivileges() // 临时

            state.memberInfo = try await info
            state.products = try await products
            state.privileges = try await privileges
        } catch {
            state.error = error.localizedDescription
        }

        state.isLoading = false
    }

    func purchase(product: CCMemberProduct) async {
        state.isLoading = true
        state.error = nil

        do {
            let info = try await purchaseMemberUseCase.execute(product: product)
            state.memberInfo = info
        } catch {
            state.error = error.localizedDescription
        }

        state.isLoading = false
    }
}
```

```swift
// Coordinators/CCMemberCoordinator.swift

/// 会员模块路由
enum CCMemberRoute: Hashable {
    case center            // 会员中心
    case purchase          // 购买页面
    case orderHistory      // 购买历史
    case privilegeDetail(id: String)  // 权益详情

/// 会员模块 Coordinator
final class CCMemberCoordinator: ObservableObject {
    @Published var path = NavigationPath()

    func navigate(to route: CCMemberRoute) {
        path.append(route)
    }

    func popToRoot() {
        path.removeLast(path.count)
    }

    @ViewBuilder
    func buildView(for route: CCMemberRoute) -> some View {
        switch route {
        case .center:
            CCMemberCenterView(coordinator: self)
        case .purchase:
            CCMemberPurchaseView(coordinator: self)
        case .orderHistory:
            Text("购买历史")
        case .privilegeDetail(let id):
            Text("权益详情: \(id)")
        }
    }
}
```

## 6. 数据流

### 6.1 购买流程

```
用户点击购买 → CCMemberPurchaseViewModel
    → CCPurchaseMemberUseCase.execute(product:)
        → CCMemberRepository.purchase(product:)
            → CCMemberRemoteDataSource.purchase(productId:)
                → APIClient.request(CCMemberAPI.purchase)
                    → 服务端处理支付
            ← CCMemberInfoDTO
        ← CCMemberInfo (DTO → Entity)
        → CCMemberLocalDataSource.cacheMemberInfo() (写入缓存)
    ← 成功/失败
→ 更新 UI 状态
```

### 6.2 会员状态校验流程

```
页面加载 → CCMemberViewModel.loadData()
    → CCFetchMemberInfoUseCase.execute()
        → CCMemberRepository.fetchMemberInfo()
            → 检查缓存 → 有 → 返回缓存
                       → 无 → CCMemberRemoteDataSource.fetchMemberInfo()
                            → APIClient.request(CCMemberAPI.info)
                            ← CCMemberInfoDTO
                        ← CCMemberInfo
                        → 写入缓存
    ← CCMemberInfo
→ 判断 isValid → true → 展示会员内容
               → false → 展示购买入口
```

## 7. 错误处理

| 场景 | 错误类型 | 用户提示 |
|------|----------|----------|
| 网络异常 | `APIError.networkFailure` | "网络连接失败，请检查网络后重试" |
| 购买失败 | `APIError.badRequest` | "购买失败，请稍后再试" |
| 支付取消 | `CancellationError` | 不提示，静默处理 |
| 恢复购买无记录 | 自定义错误 | "没有找到可恢复的购买记录" |
| 会员信息过期 | 无（正常业务逻辑） | 展示"会员已过期，续费继续享受权益" |
